package ish.template;

import java.io.*;
import java.util.*;

public class Main {
    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static StringTokenizer st;
    static int max, total;
    static ArrayList<ArrayList<Node>> link;
    static boolean[] visited;

    static class Node{ // child와 weight를 저장할 pair 정의
        int child, weight;
        Node(int child, int weight){
            this.child = child;
            this.weight = weight;
        }
    }

    public static void main(String[] args) throws IOException{
        StringBuilder sb = new StringBuilder();
        st = new StringTokenizer(br.readLine());
        max = 0;
        int parent, child, weight;

        int N = Integer.parseInt(st.nextToken());
        link = new ArrayList<ArrayList<Node>>(N+1);
        visited = new boolean[N+1];
        for(int i=0;i<=N;i++)
            link.add(new ArrayList<Node>());

        for(int i=1;i<N;i++){
            st = new StringTokenizer(br.readLine());
            parent = Integer.parseInt(st.nextToken());
            child = Integer.parseInt(st.nextToken());
            weight = Integer.parseInt(st.nextToken());

            link.get(parent).add(new Node(child, weight));
            link.get(child).add(new Node(parent, weight));
        }

        for(int i=1;i<=N;i++){
            if(link.get(i).size()==1){ // 자식 노드
                total=0;
                dfsup(i);
            }
        }


        sb.append(max);
        System.out.println(sb);
    }

    public static void dfsup(int start){
        visited[start] = true;
        if(start!=1){
            total += link.get(start).get(0).weight;
            dfsup(link.get(start).get(0).child);
            total-=link.get(start).get(0).weight;
        }
        dfsdown(start);
        visited[start] = false;
    }
    public static void dfsdown(int start){
        if(link.get(start).size()==1){ // 자식 노드
            if(total>max){
                max = total;
            }
            return;
        }
        visited[start] = true;
        ArrayList<Node> temp = link.get(start);
        for(int i=0;i<temp.size();i++){
            int child = temp.get(i).child;
            if(visited[child])
                continue;
            total+=temp.get(i).weight;
            dfsdown(child);
            total-=temp.get(i).weight;
        }
        visited[start] = false;
    }

}
